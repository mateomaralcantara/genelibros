export async function repairStep(input: {
  depth: DeepWorkDepth;
  title: string;
  sourceContent: string;
  candidate: DeepDraft;
  quality: DeepQualityScore;
  targetWords: number;
}) {
  // TOKEN_GUARD_V5.4
  // La reparación de documentos muy largos se hace por segmentos contiguos.
  // Esto impide que el modelo ignore una reducción global grande y devuelva
  // prácticamente el candidato original. Nunca reinicia etapas anteriores.

  const primaryModel = clean(
    process.env.DEEP_REPAIR_MODEL || modelFor(input.depth)
  );
  const deepModel = clean(modelFor(input.depth));

  const targetWords = Math.max(300, Number(input.targetWords || 0));
  const minWords = Math.floor(targetWords * 0.92);
  const maxWords = Math.ceil(targetWords * 1.08);
  const preferredMin = Math.round(targetWords * 0.98);
  const preferredMax = Math.round(targetWords * 1.02);

  const countWords = (value: string) =>
    clean(value).split(/\s+/).filter(Boolean).length;

  const candidateText = clean(input.candidate.content);
  const candidateWords = countWords(candidateText);

  let totalInputTokens = 0;
  let totalOutputTokens = 0;
  let totalTokens = 0;
  let calls = 0;
  const modelsUsed: string[] = [];

  const qualityRepair = JSON.stringify({
    weaknesses: input.quality?.weaknesses || [],
    requiredRepairs: input.quality?.requiredRepairs || [],
    unsupportedNewClaims: input.quality?.unsupportedNewClaims || [],
  });

  const generationConfig = (model: string, maxOutputTokens: number) => {
    const isGemini3 = /^gemini-3(?:\.|[-])/i.test(model);
    const isGemini25 = /^gemini-2\.5(?:[-]|$)/i.test(model);
    const thinkingConfig = isGemini3
      ? { thinkingLevel: "LOW" }
      : isGemini25
        ? { thinkingBudget: 1024 }
        : undefined;

    return {
      systemInstruction:
        "Eres COMPRESS_REPAIR_EDITOR V5.4 de Genelibros. " +
        "Editas texto editorial completo sin resumirlo en notas. " +
        "Preservas orden, tesis, cifras, atribuciones, procedencia, reservas y evidencia. " +
        "Eliminas redundancia y verbosidad, no sustancia.",
      temperature: isGemini3 ? 1.0 : 0.2,
      maxOutputTokens,
      ...(thinkingConfig ? { thinkingConfig } : {}),
    } as any;
  };

  const callEditor = async (args: {
    model: string;
    prompt: string;
    maxOutputTokens: number;
  }) => {
    const response = await aiClient().models.generateContent({
      model: args.model,
      contents: [{ role: "user", parts: [{ text: args.prompt }] }],
      config: generationConfig(args.model, args.maxOutputTokens),
    });

    calls += 1;
    modelsUsed.push(args.model);

    const u: AnyRow =
      (response as any)?.usageMetadata || (response as any)?.usage || {};
    const inputTokens = Number(u.promptTokenCount || u.inputTokens || 0) || 0;
    const outputTokens = Number(u.candidatesTokenCount || u.outputTokens || 0) || 0;
    const callTotalTokens =
      Number(u.totalTokenCount || u.totalTokens || inputTokens + outputTokens) || 0;

    totalInputTokens += inputTokens;
    totalOutputTokens += outputTokens;
    totalTokens += callTotalTokens;

    const text = clean((response as any)?.text);
    if (!text) {
      const reason = clean((response as any)?.candidates?.[0]?.finishReason);
      throw new Error(
        `Deep Work V5.4: ${args.model} devolvió reparación vacía; finishReason=${reason || "UNKNOWN"}.`
      );
    }
    return text;
  };

  const makeDraft = (content: string, rationale: string) => ({
    draft: {
      label: "REPAIR",
      content,
      rationale,
      strengths: [],
      risks: [],
    } as DeepDraft,
    usage: {
      inputTokens: totalInputTokens,
      outputTokens: totalOutputTokens,
      totalTokens,
      calls,
    },
    model: modelsUsed[modelsUsed.length - 1] || deepModel,
  });

  if (candidateWords > maxWords) {
    const chunkCount = Math.max(2, Math.min(4, Math.ceil(candidateWords / 1400)));
    const paragraphs = candidateText
      .split(/\n\s*\n/)
      .map((p) => clean(p))
      .filter(Boolean);

    const desiredSourcePerChunk = Math.ceil(candidateWords / chunkCount);
    const chunks: string[] = [];
    let current: string[] = [];
    let currentWords = 0;

    for (const paragraph of paragraphs) {
      const paragraphWords = countWords(paragraph);
      const canClose =
        current.length > 0 &&
        chunks.length < chunkCount - 1 &&
        currentWords + paragraphWords > desiredSourcePerChunk;

      if (canClose) {
        chunks.push(current.join("\n\n"));
        current = [];
        currentWords = 0;
      }

      current.push(paragraph);
      currentWords += paragraphWords;
    }

    if (current.length) chunks.push(current.join("\n\n"));

    if (chunks.length < 2) {
      const lines = candidateText.split(/\n+/).map(clean).filter(Boolean);
      chunks.length = 0;
      current = [];
      currentWords = 0;
      for (const line of lines) {
        const lineWords = countWords(line);
        if (
          current.length > 0 &&
          chunks.length < chunkCount - 1 &&
          currentWords + lineWords > desiredSourcePerChunk
        ) {
          chunks.push(current.join("\n"));
          current = [];
          currentWords = 0;
        }
        current.push(line);
        currentWords += lineWords;
      }
      if (current.length) chunks.push(current.join("\n"));
    }

    const compressed: string[] = [];
    let remainingTarget = targetWords;
    let remainingSource = chunks.reduce((sum, chunk) => sum + countWords(chunk), 0);

    for (let i = 0; i < chunks.length; i += 1) {
      const chunk = chunks[i];
      const sourceWords = countWords(chunk);
      const isLast = i === chunks.length - 1;
      const chunkTarget = isLast
        ? Math.max(180, remainingTarget)
        : Math.max(180, Math.round(remainingTarget * (sourceWords / remainingSource)));
      const chunkMin = Math.max(160, Math.floor(chunkTarget * 0.93));
      const chunkMax = Math.ceil(chunkTarget * 1.07);
      const deleteWords = Math.max(0, sourceWords - chunkTarget);
      const reductionPct = sourceWords
        ? Math.round((deleteWords / sourceWords) * 100)
        : 0;

      const prompt = `
CAPÍTULO: ${input.title}
SEGMENTO: ${i + 1}/${chunks.length}

Este segmento tiene aproximadamente ${sourceWords} palabras.
Debes devolver el MISMO SEGMENTO EDITADO con aproximadamente ${chunkTarget} palabras.
RANGO OBLIGATORIO DEL SEGMENTO: ${chunkMin}-${chunkMax} palabras.
Debes eliminar aproximadamente ${deleteWords} palabras (${reductionPct}% de reducción).

QUALITY REPAIRS GLOBALES:
${qualityRepair}

REGLAS:
- Conserva el orden de ideas y cualquier encabezado presente.
- Conserva hechos, cifras, atribuciones, procedencia, reservas y evidencia necesaria.
- Elimina reiteraciones, frases de transición prescindibles, duplicaciones y explicación inflada.
- No añadas hechos ni fuentes.
- No escribas resumen, notas, diagnóstico ni lista de cambios.
- No añadas introducción o conclusión que no estuvieran en este segmento.
- Devuelve EXCLUSIVAMENTE el segmento editorial completo ya comprimido.

SEGMENTO ORIGINAL:
<<<BEGIN_SEGMENT>>>
${chunk}
<<<END_SEGMENT>>>
`;

      let out = await callEditor({
        model: primaryModel,
        prompt,
        maxOutputTokens: Math.max(1200, Math.ceil(chunkTarget * 1.85)),
      });

      let outWords = countWords(out);

      if (outWords < chunkMin || outWords > chunkMax) {
        const retryPrompt = `
Tu salida anterior tuvo ${outWords} palabras y FALLÓ el rango ${chunkMin}-${chunkMax}.
Repite la edición DESDE EL SEGMENTO ORIGINAL. No continúes ni corrijas la salida fallida.
Debes producir entre ${chunkMin} y ${chunkMax} palabras.

QUALITY REPAIRS:
${qualityRepair}

SEGMENTO ORIGINAL:
<<<BEGIN_SEGMENT>>>
${chunk}
<<<END_SEGMENT>>>

Devuelve exclusivamente el segmento completo comprimido. Sin comentarios.
`;
        out = await callEditor({
          model: deepModel,
          prompt: retryPrompt,
          maxOutputTokens: Math.max(1200, Math.ceil(chunkTarget * 1.9)),
        });
        outWords = countWords(out);
      }

      compressed.push(out);
      remainingTarget = Math.max(0, remainingTarget - outWords);
      remainingSource = Math.max(1, remainingSource - sourceWords);
    }

    let finalText = clean(compressed.join("\n\n"));
    let finalWords = countWords(finalText);

    if (finalWords >= minWords && finalWords <= maxWords) {
      return makeDraft(
        finalText,
        `V5.4 chunked repair; source=${candidateWords}; words=${finalWords}; chunks=${chunks.length}`
      );
    }

    const calibrationPrompt = `
CAPÍTULO: ${input.title}

El texto ya fue comprimido por segmentos y ahora tiene ${finalWords} palabras.
Objetivo: ${targetWords} palabras.
RANGO DURO: ${minWords}-${maxWords} palabras.
RANGO PREFERIDO: ${preferredMin}-${preferredMax} palabras.

Haz únicamente una calibración editorial de longitud y continuidad.
No restaures material eliminado. No añadas hechos. Conserva cifras, atribuciones,
procedencia, reservas y evidencia. Devuelve exclusivamente el capítulo completo.

TEXTO YA COMPRIMIDO:
<<<BEGIN_COMPRESSED>>>
${finalText}
<<<END_COMPRESSED>>>
`;

    finalText = await callEditor({
      model: deepModel,
      prompt: calibrationPrompt,
      maxOutputTokens: Math.max(5000, Math.ceil(targetWords * 1.9)),
    });
    finalWords = countWords(finalText);

    if (finalWords >= minWords && finalWords <= maxWords) {
      return makeDraft(
        finalText,
        `V5.4 chunked+calibrated repair; source=${candidateWords}; words=${finalWords}; chunks=${chunks.length}`
      );
    }

    throw new Error(
      `Deep Work V5.4: compresión segmentada fuera de rango (${finalWords}; requerido ${minWords}-${maxWords}); source=${candidateWords}; chunks=${chunks.length}; model=${deepModel}; preview=${JSON.stringify(finalText.slice(0, 700))}`
    );
  }

  let lastText = "";
  let lastWordCount = 0;

  for (let attempt = 1; attempt <= 2; attempt += 1) {
    const model = attempt === 1 ? primaryModel : deepModel;
    const prompt = `
CAPÍTULO: ${input.title}
OBJETIVO: ${targetWords} palabras.
RANGO OBLIGATORIO: ${minWords}-${maxWords} palabras.

QUALITY REPAIRS:
${qualityRepair}

CANDIDATO:
<<<BEGIN_CANDIDATE>>>
${candidateText}
<<<END_CANDIDATE>>>

Repara los defectos sin inventar hechos ni fuentes. Conserva evidencia, atribuciones,
procedencia y reservas. Devuelve exclusivamente el capítulo completo reparado.
`;

    lastText = await callEditor({
      model,
      prompt,
      maxOutputTokens: Math.max(6000, Math.ceil(targetWords * 2.1)),
    });
    lastWordCount = countWords(lastText);

    if (lastWordCount >= minWords && lastWordCount <= maxWords) {
      return makeDraft(
        lastText,
        `V5.4 full-document repair; attempt=${attempt}; words=${lastWordCount}`
      );
    }
  }

  throw new Error(
    `Deep Work V5.4: reparación fuera de rango (${lastWordCount}; requerido ${minWords}-${maxWords}); model=${deepModel}; preview=${JSON.stringify(lastText.slice(0, 700))}`
  );
}
